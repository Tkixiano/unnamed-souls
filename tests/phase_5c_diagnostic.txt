var t=get_tree()
var a=t.root.get_node("AudioDirector")
var cue=a.catalog.find(&"player_light")
var wav=AudioStreamWAV.new()
wav.format=AudioStreamWAV.FORMAT_16_BITS
wav.mix_rate=8000
var bytes=PackedByteArray()
bytes.resize(128000)
bytes.fill(0)
wav.data=bytes
cue.stream=wav
var marker=Node3D.new()
t.current_scene.add_child(marker)
a.play_event(&"player_light",marker,"diagnostic")
await t.create_timer(0.2,true).timeout
var v=a._spatial.filter(func(x):return x.playing)[0]
t.paused=true
for i in range(4):await t.process_frame
var before={"paused":v.stream_paused,"position":v.get_playback_position(),"mode":v.process_mode,"can_process":v.can_process()}
await t.create_timer(0.25,true).timeout
var after={"paused":v.stream_paused,"position":v.get_playback_position(),"mode":v.process_mode,"can_process":v.can_process()}
t.paused=false
a.stop_effects()
cue.stream=null
marker.queue_free()
var r={"before":before,"after":after}
FileAccess.open("res://tests/phase_5c_pause_diagnostic.json",FileAccess.WRITE).store_string(JSON.stringify(r,"\t"))
return r
